    private func scheduleMicrophoneStartupRetry(
        sessionID: UUID,
        attempt: Int,
        error: Error,
        onReady: @escaping () -> Void,
        onFailure: @escaping (Error) -> Void
    ) {
        guard activeSessionID == sessionID, isRecording, !isSystemSleeping else { return }
        guard attempt < microphoneStartupRetryLimit else {
            audioCaptureState = .failed(error.localizedDescription)
            onFailure(error)
            return
        }

        status = "正在准备麦克风…"
        audioCaptureState = .recovering
        isRecoveringAudioCapture = true
        let workItem = DispatchWorkItem { [weak self] in
            self?.startMicrophoneCaptureWithRetry(
                sessionID: sessionID,
                attempt: attempt + 1,
                onReady: onReady,
                onFailure: onFailure
            )
        }
        microphoneStartupWorkItem?.cancel()
        microphoneStartupWorkItem = workItem
        DispatchQueue.main.asyncAfter(deadline: .now() + microphoneStartupRetryDelay, execute: workItem)
    }
